/* record any newly unlocked achievements and tell the player; the first check after this update just records what was already earned */
function achCheck(){
  try{
    const s=ST(),list=achList(),fresh=[];
    list.forEach(a=>{if(a.val()>=a.goal&&!s.ach[a.id]){s.ach[a.id]=Date.now();fresh.push(a)}});
    if(!fresh.length)return;
    saveP();
    if(!s.achInit){s.achInit=1;saveP();return}
    const med=a=>MEDALS[a.med][0];
    if(fresh.length<=2)fresh.forEach(a=>toast('🏅 <span><b>Achievement unlocked</b> · '+escH(a.name)+' <small>('+med(a)+')</small></span>',3600));
    else toast('🏅 <span><b>'+fresh.length+' achievements unlocked</b> · open Badges to see them</span>',4200);
  }catch(e){}
}
